cmake_minimum_required(VERSION 3.24)
project(CGH_LAPI VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CGH-LAPI requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CGH_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../Cpp/include" CACHE PATH "Directory containing cgh_sdk.h")
add_library(cgh_lapi_win64 SHARED src/core.cpp)
target_include_directories(cgh_lapi_win64 PRIVATE "${CGH_SDK_DIR}")
target_compile_definitions(cgh_lapi_win64 PRIVATE LAPI_VERSION="${PROJECT_VERSION}")
add_library(gmsv_cgh_lapi_win64 SHARED src/glua.cpp)
add_library(gmcl_cgh_lapi_win64 SHARED src/glua.cpp)
target_compile_definitions(gmcl_cgh_lapi_win64 PRIVATE LAPI_CLIENT=1)

include(CTest)
if(BUILD_TESTING)
  add_executable(lapi_tests tests/core_tests.cpp)
  target_include_directories(lapi_tests PRIVATE src "${CGH_SDK_DIR}")
  add_dependencies(lapi_tests cgh_lapi_win64)
  add_test(NAME lapi_core COMMAND lapi_tests "$<TARGET_FILE:cgh_lapi_win64>")
endif()
foreach(target cgh_lapi_win64 gmsv_cgh_lapi_win64 gmcl_cgh_lapi_win64 lapi_tests)
  if(TARGET ${target})
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
    endif()
  endif()
endforeach()

install(TARGETS cgh_lapi_win64 RUNTIME DESTINATION Mods/CGH-LAPI/Native)
install(FILES cghm_metadata.json DESTINATION Mods/CGH-LAPI)
install(TARGETS gmsv_cgh_lapi_win64 gmcl_cgh_lapi_win64 RUNTIME DESTINATION GMod/garrysmod/lua/bin)
install(FILES lua/cgh_lapi.lua DESTINATION GMod/garrysmod/lua/autorun)
