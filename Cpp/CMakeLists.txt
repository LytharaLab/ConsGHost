cmake_minimum_required(VERSION 3.24)
project(ConsGHost VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CGH requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED DOTNET_NATIVE_DIR)
  file(GLOB _dotnet_dirs "C:/Program Files/dotnet/packs/Microsoft.NETCore.App.Host.win-x64/10.*/runtimes/win-x64/native")
  list(SORT _dotnet_dirs COMPARE NATURAL ORDER DESCENDING)
  list(GET _dotnet_dirs 0 DOTNET_NATIVE_DIR)
endif()
if(NOT EXISTS "${DOTNET_NATIVE_DIR}/libnethost.lib")
  message(FATAL_ERROR "Set DOTNET_NATIVE_DIR to the .NET 10 win-x64 SDK native hosting pack")
endif()
add_subdirectory(third_party/minhook-1.3.4 EXCLUDE_FROM_ALL)
add_library(cgh_runtime_win64 SHARED src/runtime.cpp src/hosting.cpp)
target_include_directories(cgh_runtime_win64 PRIVATE include "${DOTNET_NATIVE_DIR}")
target_compile_definitions(cgh_runtime_win64 PRIVATE NETHOST_USE_AS_STATIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(cgh_runtime_win64 PRIVATE minhook "${DOTNET_NATIVE_DIR}/libnethost.lib" advapi32)
add_executable(cgh_mrt_win64 src/mrt.cpp)
target_include_directories(cgh_mrt_win64 PRIVATE include)
target_compile_definitions(cgh_mrt_win64 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(cgh_example_native SHARED examples/example.cpp)
target_include_directories(cgh_example_native PRIVATE include)
add_executable(cgh_test_game tests/test_game.cpp)
target_compile_definitions(cgh_test_game PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(cgh_test_extra SHARED tests/extra.cpp)
add_library(cgh_test_server SHARED tests/server.cpp)
target_include_directories(cgh_test_server PRIVATE include)
set_target_properties(cgh_test_server PROPERTIES OUTPUT_NAME server)
foreach(target cgh_runtime_win64 cgh_mrt_win64 cgh_example_native cgh_test_game cgh_test_extra cgh_test_server)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
  endif()
endforeach()
enable_testing()
add_test(NAME hook_and_bridge COMMAND cgh_mrt_win64 --native-test)
