cmake_minimum_required(VERSION 3.24)
set(CGH_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CGH_VERSION_FILE}")
file(READ "${CGH_VERSION_FILE}" CGH_PRODUCT_VERSION)
string(STRIP "${CGH_PRODUCT_VERSION}" CGH_PRODUCT_VERSION)
if(NOT CGH_PRODUCT_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain a stable major.minor.patch version with no leading zeros")
endif()
if(CMAKE_MATCH_1 GREATER 65534 OR CMAKE_MATCH_2 GREATER 65534 OR CMAKE_MATCH_3 GREATER 65534)
  message(FATAL_ERROR "VERSION components must be at most 65534")
endif()
project(ConsGHost VERSION ${CGH_PRODUCT_VERSION} LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CGH requires Windows x64")
endif()
enable_language(RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED DOTNET_NATIVE_DIR)
  file(GLOB _dotnet_dirs "C:/Program Files/dotnet/packs/Microsoft.NETCore.App.Host.win-x64/10.*/runtimes/win-x64/native")
  list(SORT _dotnet_dirs COMPARE NATURAL ORDER DESCENDING)
  list(GET _dotnet_dirs 0 DOTNET_NATIVE_DIR)
endif()
if(NOT EXISTS "${DOTNET_NATIVE_DIR}/libnethost.lib")
  message(FATAL_ERROR "Set DOTNET_NATIVE_DIR to the .NET 10 win-x64 SDK native hosting pack")
endif()
add_subdirectory(third_party/minhook-1.3.4 EXCLUDE_FROM_ALL)
add_library(cgh_runtime_win64 SHARED src/runtime.cpp src/hosting.cpp)
target_include_directories(cgh_runtime_win64 PRIVATE include "${DOTNET_NATIVE_DIR}")
target_compile_definitions(cgh_runtime_win64 PRIVATE NETHOST_USE_AS_STATIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(cgh_runtime_win64 PRIVATE minhook "${DOTNET_NATIVE_DIR}/libnethost.lib" advapi32)
add_executable(cgh_mrt_win64 src/mrt.cpp)
target_include_directories(cgh_mrt_win64 PRIVATE include)
target_compile_definitions(cgh_mrt_win64 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN CGH_PRODUCT_VERSION="${PROJECT_VERSION}")
function(cgh_version_resource target filename file_type description)
  set(CGH_ORIGINAL_FILENAME "${filename}")
  set(CGH_FILE_TYPE "${file_type}")
  set(CGH_FILE_DESCRIPTION "${description}")
  set(resource "${CMAKE_CURRENT_BINARY_DIR}/${target}_version.rc")
  configure_file(src/version.rc.in "${resource}" @ONLY)
  target_sources(${target} PRIVATE "${resource}")
endfunction()
cgh_version_resource(cgh_mrt_win64 cgh_mrt_win64.exe VFT_APP "ConsGHost native launcher")
cgh_version_resource(cgh_runtime_win64 cgh_runtime_win64.dll VFT_DLL "ConsGHost native runtime")
add_library(cgh_example_native SHARED examples/example.cpp)
target_include_directories(cgh_example_native PRIVATE include)

add_library(cgh_test_extra SHARED tests/extra.cpp)

foreach(target cgh_runtime_win64 cgh_mrt_win64 cgh_example_native cgh_test_game cgh_test_extra cgh_test_server)
